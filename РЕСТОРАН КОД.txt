(function() {
  const grid = document.getElementById('rs-menu-grid');
  const tabsEl = document.getElementById('rs-tabs');
  let activeCat = MENU[0].category;


  function fmtPrice(p) {
    return p.toLocaleString('ru-RU') + '<span class="currency">' + RESTAURANT.currency + '</span>';
  }


  function renderTabs() {
    tabsEl.innerHTML = '';
    MENU.forEach(cat => {
      const btn = document.createElement('button');
      btn.className = 'rs-tab' + (cat.category === activeCat ? ' active' : '');
      btn.textContent = cat.category;
      btn.onclick = () => { activeCat = cat.category; renderTabs(); renderDishes(); };
      tabsEl.appendChild(btn);
    });
  }


  function renderDishes() {
    grid.innerHTML = '';
    const cat = MENU.find(c => c.category === activeCat);
    if (!cat) return;
    cat.items.forEach((dish, i) => {
      const card = document.createElement('article');
      card.className = 'rs-dish';
      card.style.animationDelay = (i * 0.08) + 's';
      const badge = dish.badge ? '<div class="rs-dish-badge">' + dish.badge + '</div>' : '';
      card.innerHTML =
        '<div class="rs-dish-img-wrap">' +
          badge +
          '<img class="rs-dish-img" src="' + dish.photo + '" alt="' + dish.name + '" loading="lazy" onerror="this.src=\'https://images.unsplash.com/photo-1414235077428-338989a2e8c0?w=800&q=85\'">' +
        '</div>' +
        '<div class="rs-dish-body">' +
          '<div class="rs-dish-head">' +
            '<h3 class="rs-dish-name">' + dish.name + '</h3>' +
            '<div class="rs-dish-price">' + fmtPrice(dish.price) + '</div>' +
          '</div>' +
          '<p class="rs-dish-desc">' + dish.desc + '</p>' +
          (dish.weight ? '<div class="rs-dish-weight">' + dish.weight + '</div>' : '') +
        '</div>';
      grid.appendChild(card);
    });
  }


  function renderGallery() {
    const g = document.getElementById('rs-gallery');
    GALLERY.forEach(src => {
      const img = document.createElement('img');
      img.src = src;
      img.loading = 'lazy';
      g.appendChild(img);
    });
  }


  function renderReserve() {
    document.getElementById('rs-reserve-phone').href = 'tel:' + RESTAURANT.phoneLink;
    document.getElementById('rs-reserve-phone').textContent = RESTAURANT.phone;
    document.getElementById('rs-reserve-info').innerHTML =
      '<div class="rs-reserve-info-item"><div class="label">Адрес</div><div class="value">' + RESTAURANT.address + '</div></div>' +
      '<div class="rs-reserve-info-item"><div class="label">Часы работы</div><div class="value">' + RESTAURANT.hours + '</div></div>' +
      '<div class="rs-reserve-info-item"><div class="label">Телефон</div><div class="value">' + RESTAURANT.phone + '</div></div>';
    document.getElementById('rs-footer-name').textContent = RESTAURANT.name.toUpperCase();
  }


  renderTabs();
  renderDishes();
  renderGallery();
  renderReserve();
})();
</script>
